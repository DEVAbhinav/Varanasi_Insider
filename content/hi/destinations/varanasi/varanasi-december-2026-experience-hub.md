  document.addEventListener('DOMContentLoaded', function () {
    document.querySelectorAll('[data-december-tabs]').forEach(function (group) {
      const buttons = group.querySelectorAll('.december-tab');
      const panels = group.querySelectorAll('[data-december-panel]');
      buttons.forEach(function (button) {
        button.addEventListener('click', function () {
          const target = button.getAttribute('data-december-target');
          buttons.forEach((b) => b.classList.remove('is-active'));
          panels.forEach((panel) => panel.classList.remove('is-active'));
          button.classList.add('is-active');
          const activePanel = group.querySelector('[data-december-panel="' + target + '"]');
          if (activePanel) {
            activePanel.classList.add('is-active');
          }
        });
      });
    });
  });
</script>

## 🧭 टैक्सी + अनुभव कॉम्बो

| परिदृश्य | जोड़ें | टैक्सी मूव |
| --- | --- | --- |
| **सांस्कृतिक हफ्ता (2-15 Dec)** | संगमम + सिल्क | BHU ➝ ठठेरी ➝ असी लूप (एयरपोर्ट गाइड CTA) |
| **सूर्योदय फोकस** | योग + कबीर डॉन + मलाईयो | 4:45 AM पिकअप, कैब असी पर वेट, फिर गोडोलिया ड्रॉप |
| **वीकेंड फेस्टिवल (19-21 Dec)** | कबीर + सिल्क + फूड | Delegate पास + शॉपिंग कैब + पोस्ट-फेस्ट Ayodhya/Prayagraj रन |

## 🧵 सैंपल सप्ताह

1. **Day 1:** VNS आगमन, असी में चेक-इन, शाम संगमम वॉक।
2. **Day 2:** सूर्योदय योग ➝ मलाईयो ➝ BHU सेशन।
3. **Day 3:** दोपहर सारनाथ दौरा ➝ सिल्क शॉपिंग ➝ कबीर वॉलंटियर ब्रीफ़।
4. **Day 4:** कबीर सूर्योदय ➝ आराम ➝ शाम कॉन्सर्ट।
5. **Day 5:** प्रयागराज या अयोध्या सड़क यात्रा।
6. **Day 6:** सुबह मुक्त, स्ट्रीट फूड, पैक-आउट।

इस हब को बुकमार्क करें और हर टैब से एक-एक अनुभव चुनकर दिसंबर को स्ट्रक्चर्ड फिर भी सहज रखें।
